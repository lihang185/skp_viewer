cmake_minimum_required(VERSION 3.15)
project(skp_viewer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(skp_archive STATIC src/skp_archive.cpp)
target_include_directories(skp_archive PUBLIC include)
file(GLOB LIBTESS2_SOURCES CONFIGURE_DEPENDS third_party/libtess2/Source/*.c)
add_library(libtess2 STATIC ${LIBTESS2_SOURCES})
target_include_directories(libtess2 PUBLIC third_party/libtess2/Include)
add_library(skp_model STATIC src/legacy_reader.cpp src/skp_model.cpp src/skp_triangulate.cpp)
target_link_libraries(skp_model PUBLIC libtess2)
target_include_directories(skp_model PUBLIC include)
set_target_properties(skp_model PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(slapi_compat SHARED src/slapi_compat.cpp src/skp_mesh_api.cpp)
target_compile_definitions(slapi_compat PRIVATE SLAPI_COMPAT_EXPORTS)
target_link_libraries(slapi_compat PRIVATE skp_model)
target_include_directories(slapi_compat PUBLIC include)
set_target_properties(slapi_compat PROPERTIES OUTPUT_NAME slapi)
add_executable(read_model tools/read_model.cpp)
target_link_libraries(read_model PRIVATE slapi_compat)
add_executable(compat_api_test tests/compat_api_test.cpp src/slapi_compat.cpp)
target_include_directories(compat_api_test PRIVATE include)
add_executable(model_parser_test tests/model_parser_test.cpp)
target_link_libraries(model_parser_test PRIVATE skp_model)
add_executable(skp_inspect tools/inspect_skp.cpp)
target_link_libraries(skp_inspect PRIVATE skp_archive skp_model)
enable_testing()
add_executable(triangulate_test tests/triangulate_test.cpp)
target_link_libraries(triangulate_test PRIVATE skp_model)
add_test(NAME triangulation COMMAND triangulate_test)
add_executable(mesh_api_test tests/mesh_api_test.cpp)
add_executable(texture_api_test tests/texture_api_test.cpp)
target_link_libraries(texture_api_test PRIVATE slapi_compat)
add_test(NAME texture_api COMMAND texture_api_test)
target_link_libraries(mesh_api_test PRIVATE slapi_compat)
add_test(NAME mesh_api COMMAND mesh_api_test "${CMAKE_CURRENT_SOURCE_DIR}/test2_box.skp")
file(GLOB DATA_MODELS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/*.skp")
foreach(MODEL_PATH IN LISTS DATA_MODELS)
  get_filename_component(MODEL_NAME "${MODEL_PATH}" NAME_WE)
  add_test(NAME "triangulate_data_${MODEL_NAME}" COMMAND triangulate_test "${MODEL_PATH}")
  set_tests_properties("triangulate_data_${MODEL_NAME}" PROPERTIES TIMEOUT 90)
endforeach()
add_test(NAME compat_api COMMAND compat_api_test)
add_test(NAME public_api_read COMMAND read_model "${CMAKE_CURRENT_SOURCE_DIR}/test2_box.skp")
add_test(NAME model_parser COMMAND model_parser_test "${CMAKE_CURRENT_SOURCE_DIR}/test2_box.skp" "${CMAKE_CURRENT_BINARY_DIR}/parser_test_input.skp")
add_executable(archive_test tests/archive_test.cpp)
target_link_libraries(archive_test PRIVATE skp_archive)
add_test(NAME archive_header COMMAND archive_test "${CMAKE_CURRENT_SOURCE_DIR}/test2_box.skp")
if(WIN32)
  target_sources(slapi_compat PRIVATE src/slapi_exports.def)
endif()
install(TARGETS slapi_compat read_model RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/slapi_compat.h DESTINATION include)
install(FILES include/skp_mesh_api.h DESTINATION include)
install(FILES third_party/libtess2/LICENSE.txt DESTINATION licenses/libtess2)
install(FILES tools/read_model.cpp DESTINATION examples)
